'------------------------------ layout helpers ---------------------------------
Private Function AddCtl(ByVal progId As String, ByVal l As Single, ByVal t As Single, _
                        ByVal w As Single, ByVal h As Single, Optional ByVal cap As String = "", _
                        Optional ByVal tip As String = "") As Object
    Dim c As Object
    mCtlCount = mCtlCount + 1
    Set c = Me.Controls.Add(progId, "rt" & mCtlCount, True)   ' unique name for every control
    c.Left = l
    c.Top = t
    c.Width = w
    c.Height = h
    If cap <> "" Then c.Caption = cap
    If tip <> "" Then c.ControlTipText = tip
    Set AddCtl = c
End Function

Private Sub Lbl(ByVal cap As String, ByVal l As Single, ByVal t As Single, Optional ByVal w As Single = 80, _
                Optional ByVal tip As String = "")
    AddCtl "Forms.Label.1", l, t + 2, w, 12, cap, tip
End Sub

'------------------------------ modes ------------------------------------------
Private Sub FillModes()
    Dim names As Collection, ids As Collection, i As Long, t As Long
    Set names = New Collection
    Set ids = New Collection
    If Not mDoc Is Nothing Then t = mDoc.GetType
    Select Case t
        Case swDocumentTypes_e.swDocPART
            names.Add "Bodies (each visible body)": ids.Add MODE_BODIES
            names.Add "Configurations": ids.Add MODE_CONFIGS
            names.Add "Configurations x bodies": ids.Add MODE_CFGBODIES
        Case swDocumentTypes_e.swDocASSEMBLY
            names.Add "Configurations": ids.Add MODE_CONFIGS
            names.Add "Components (each visible part)": ids.Add MODE_COMPONENTS
            names.Add "Components x bodies": ids.Add MODE_COMPBODIES
        Case swDocumentTypes_e.swDocDRAWING
            names.Add "Drawing sheets": ids.Add MODE_SHEETS
    End Select
    names.Add "Batch (folder or files)": ids.Add MODE_BATCH
    ReDim mModeIds(0 To ids.Count - 1)
    cboMode.Clear
    For i = 1 To ids.Count
        cboMode.AddItem names(i)
        mModeIds(i - 1) = ids(i)
    Next i
    cboMode.ListIndex = 0          ' fires cboMode_Change
End Sub

Private Function CurMode() As Long
    If cboMode.ListIndex < 0 Then
        CurMode = MODE_BATCH
    Else
        CurMode = mModeIds(cboMode.ListIndex)
    End If
End Function

Private Sub cboMode_Change()
    If cboMode.ListIndex < 0 Then Exit Sub
    FillItems
    FillFormats
End Sub

Private Sub chkSurf_Click()
    If CurMode() = MODE_BODIES Then FillItems
End Sub

'------------------------------ item list --------------------------------------
Private Sub ClearItems()
    lstItems.Clear
    Set mKeys = New Collection
End Sub

Private Sub AddListItem(ByVal text As String, ByVal key As String, ByVal ticked As Boolean)
    lstItems.AddItem text
    mKeys.Add key
    lstItems.Selected(lstItems.ListCount - 1) = ticked
End Sub

Private Sub SetBatchEnabled(ByVal isOn As Boolean)
    txtBatch.Enabled = isOn
    btnBrowseBatch.Enabled = isOn
    btnRefresh.Enabled = isOn
    btnAddFiles.Enabled = isOn
    chkRecurse.Enabled = isOn
    cboBatchModel.Enabled = isOn
    txtCfgFilter.Enabled = isOn
End Sub

Private Sub FillItems()
    Dim v As Variant, i As Long, c As Collection, activeName As String
    Dim drw As SldWorks.DrawingDoc, md As Long, d As Object, k As Variant
    md = CurMode()
    ClearItems
    chkSurf.Enabled = (md = MODE_BODIES Or md = MODE_CFGBODIES Or md = MODE_COMPBODIES Or md = MODE_BATCH)
    SetBatchEnabled (md = MODE_BATCH)
    Select Case md
        Case MODE_BODIES
            lblItems.Caption = "Visible bodies:"
            Set c = GetBodyNames(mDoc, CBool(chkSurf.Value))
            For i = 1 To c.Count
                AddListItem c(i), c(i), True
            Next i
        Case MODE_CONFIGS, MODE_CFGBODIES
            If md = MODE_CONFIGS Then
                lblItems.Caption = "Configurations:"
            Else
                lblItems.Caption = "Configurations (all visible bodies of each):"
            End If
            activeName = mDoc.ConfigurationManager.ActiveConfiguration.Name
            v = mDoc.GetConfigurationNames
            If IsArray(v) Then
                For i = LBound(v) To UBound(v)
                    AddListItem CStr(v(i)), CStr(v(i)), (CStr(v(i)) = activeName)
                Next i
            End If
        Case MODE_COMPONENTS, MODE_COMPBODIES
            lblItems.Caption = "Parts [configuration] (each once):"
            Set d = AssemblyParts(mDoc, True)
            For Each k In d.Keys
                AddListItem PartKeyDisplay(CStr(k)), CStr(k), True
            Next k
        Case MODE_SHEETS
            lblItems.Caption = "Sheets:"
            Set drw = mDoc
            v = drw.GetSheetNames
            If IsArray(v) Then
                For i = LBound(v) To UBound(v)
                    AddListItem CStr(v(i)), CStr(v(i)), True
                Next i
            End If
        Case MODE_BATCH
            lblItems.Caption = "Files:"
            If txtBatch.Text = "" And Not mDoc Is Nothing Then
                If mDoc.GetPathName <> "" Then
                    txtBatch.Text = Left$(mDoc.GetPathName, InStrRev(mDoc.GetPathName, "\") - 1)
                End If
            End If
            If txtBatch.Text = "" Then txtBatch.Text = swApp.GetCurrentWorkingDirectory
            Set c = ListModelFiles(txtBatch.Text, CBool(chkRecurse.Value))
            For i = 1 To c.Count
                AddListItem c(i), c(i), True
            Next i
    End Select
End Sub

Private Sub FillFormats()
    Dim i As Long, n As Long, wantKind As String, prefix As String
    Select Case CurMode()
        Case MODE_SHEETS: wantKind = "DRAWING"
        Case MODE_BATCH: wantKind = ""
        Case Else: wantKind = "MODEL"
    End Select
    lstFormats.Clear
    ReDim mFmtIdx(0 To gFormatCount)
    For i = 0 To gFormatCount - 1
        If wantKind = "" Or gFormats(i).Kind = wantKind Then
            prefix = ""
            If wantKind = "" Then
                If gFormats(i).Kind = "MODEL" Then prefix = "Model: " Else prefix = "Drawing: "
            End If
            lstFormats.AddItem prefix & gFormats(i).Label & "  (" & gFormats(i).Ext & ")"
            mFmtIdx(n) = i
            lstFormats.Selected(n) = gFormats(i).DefaultOn
            n = n + 1
        End If
    Next i
End Sub

Private Sub SetMatches(ByVal v As Boolean)
    Dim i As Long
    If txtFilter.Text = "" Then Exit Sub
    For i = 0 To lstItems.ListCount - 1
        If MatchesFilter(CStr(lstItems.List(i)), txtFilter.Text, CBool(chkCase.Value)) Then lstItems.Selected(i) = v
    Next i
End Sub

'------------------------------ folders / files --------------------------------
Private Sub cboOut_Change()
    Select Case cboOut.ListIndex
        Case 0
            txtOut.Text = ""
            txtOut.Enabled = False
        Case 1
            txtOut.Enabled = True
            If txtOut.Text = "" Or InStr(txtOut.Text, "\") > 0 Then txtOut.Text = "{file}_export"
        Case 2
            txtOut.Enabled = True
            If InStr(txtOut.Text, "\") = 0 Then txtOut.Text = ""
    End Select
    btnBrowseOut.Enabled = (cboOut.ListIndex = 2)
End Sub

Private Sub btnBrowseOut_Click()
    Dim p As String
    p = BrowseForFolder("Output folder:")
    If p <> "" Then txtOut.Text = p
End Sub

Private Sub btnBrowseBatch_Click()
    Dim p As String
    p = BrowseForFolder("Folder to batch export:")
    If p <> "" Then
        txtBatch.Text = p
        If CurMode() = MODE_BATCH Then FillItems
    End If
End Sub

Private Sub chkRecurse_Click()

End Sub

Private Sub btnAddFiles_Click()
    Dim c As Collection, i As Long, j As Long, dup As Boolean, n As Long
    If CurMode() <> MODE_BATCH Then Exit Sub
    Set c = PickFiles("Add files to the batch", txtBatch.Text)
    For i = 1 To c.Count
        dup = False
        For j = 1 To mKeys.Count
            If StrComp(CStr(mKeys(j)), CStr(c(i)), vbTextCompare) = 0 Then
                dup = True
                lstItems.Selected(j - 1) = True
                Exit For
            End If
        Next j
        If Not dup Then
            AddListItem c(i), c(i), True
            n = n + 1
        End If
    Next i
    If c.Count > 0 Then ShowStatus n & " file(s) added."
End Sub

'------------------------------ run --------------------------------------------
Public Sub ShowStatus(ByVal s As String)
    On Error Resume Next
    lblStatus.Caption = s
    Me.Repaint
End Sub

Private Sub btnClose_Click()

        gCancel = True
        ShowStatus "Cancelling after the current file..."
        Exit Sub
    End If
    gFormLoaded = False
    Unload Me
End Sub

Private Sub btnExport_Click()
    Dim i As Long, anyFmt As Boolean
    If mBusy Then Exit Sub

    Set gSelItems = New Collection
    For i = 0 To lstItems.ListCount - 1
        If lstItems.Selected(i) Then gSelItems.Add CStr(mKeys(i + 1))
    Next i
    If gSelItems.Count = 0 Then
        MsgBox "Tick at least one item.", vbExclamation, "SWExporter"
        Exit Sub
    End If

    For i = 0 To gFormatCount - 1
        gFormats(i).Selected = False
    Next i
    For i = 0 To lstFormats.ListCount - 1
        If lstFormats.Selected(i) Then
            gFormats(mFmtIdx(i)).Selected = True
            anyFmt = True
        End If
    Next i
    If Not anyFmt Then
        MsgBox "Tick at least one format.", vbExclamation, "SWExporter"
        Exit Sub
    End If

    gMode = CurMode()
    gIncludeSurfaces = CBool(chkSurf.Value)
    gOutMode = cboOut.ListIndex
    gOutText = Trim$(txtOut.Text)
    If gOutMode = 2 And gOutText = "" Then
        MsgBox "Choose a custom output folder.", vbExclamation, "SWExporter"
        Exit Sub
    End If
    gTemplate = Trim$(txtTemplate.Text)
    gCollision = cboCollision.ListIndex
    gStlUnits = cboUnits.ListIndex                  ' 0..4 = swLengthUnit_e mm, cm, m, in, ft
    gStlQuality = Choose(cboQuality.ListIndex + 1, swSTLQuality_e.swSTLQuality_Coarse, _
                         swSTLQuality_e.swSTLQuality_Fine, swSTLQuality_e.swSTLQuality_Custom)
    gStlBinary = CBool(chkBinary.Value)
    gPdfCombined = CBool(chkPdfOne.Value)
    gDryRun = CBool(chkDry.Value)
    gWriteLog = CBool(chkLog.Value)
    gBatchFolder = Trim$(txtBatch.Text)
    gBatchModelMode = Choose(cboBatchModel.ListIndex + 1, MODE_CONFIGS, MODE_BODIES, MODE_CFGBODIES)
    gCfgFilter = Trim$(txtCfgFilter.Text)
    gCfgFilterCase = CBool(chkCase.Value)

    If gMode = MODE_BATCH And Not gDryRun Then
        If MsgBox("Export " & gSelItems.Count & " file(s)? Files that are not already open will be " & _
                  "opened and closed again without saving.", vbOKCancel + vbQuestion, "SWExporter") <> vbOK Then Exit Sub
    End If

    mBusy = True
    btnExport.Enabled = False
    btnClose.Caption = "Cancel"
    RunExport mDoc
    mBusy = False
    btnExport.Enabled = True
    btnClose.Caption = "Close"
    ShowStatus "Done."
End Sub
