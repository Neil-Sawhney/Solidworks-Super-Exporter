'=============================== LAYOUT HELPERS ================================
' Every control gets a unique name; creation order = tab order.
Private Function AddCtl(ByVal progId As String, ByVal l As Single, ByVal t As Single, _
                        ByVal w As Single, ByVal h As Single, Optional ByVal cap As String = "", _
                        Optional ByVal tip As String = "") As Object
    Dim c As Object
    mCtlCount = mCtlCount + 1
    Set c = Me.Controls.Add(progId, "rt" & mCtlCount, True)
    c.Left = l
    c.Top = t
    c.Width = w
    c.Height = h
    If cap <> "" Or progId = "Forms.Label.1" Then c.Caption = cap   ' labels: never a default caption
    If tip <> "" Then c.ControlTipText = tip
    Set AddCtl = c
End Function

' Caption in front of a field (2 pt lower so the text lines up with the field).
Private Function Lbl(ByVal cap As String, ByVal l As Single, ByVal t As Single, _
                     Optional ByVal w As Single = LBL_W, Optional ByVal tip As String = "") As MSForms.Label
    Set Lbl = AddCtl("Forms.Label.1", l, t + 2, w, 12, cap, tip)
End Function

Private Function Btn(ByVal cap As String, ByVal l As Single, ByVal t As Single, ByVal w As Single, _
                     ByVal tip As String) As MSForms.CommandButton
    Set Btn = AddCtl("Forms.CommandButton.1", l, t, w, 18, cap, tip)
End Function

Private Function Chk(ByVal cap As String, ByVal l As Single, ByVal t As Single, ByVal w As Single, _
                     ByVal tip As String) As MSForms.CheckBox

Private Function Txt(ByVal l As Single, ByVal t As Single, ByVal w As Single, ByVal tip As String) As MSForms.TextBox
    Set Txt = AddCtl("Forms.TextBox.1", l, t, w, ROW_H, , tip)
End Function

' Drop-down list (no typing) filled with items.
Private Function Combo(ByVal l As Single, ByVal t As Single, ByVal w As Single, ByVal tip As String, _
                       ByVal items As Variant) As MSForms.ComboBox
    Dim c As MSForms.ComboBox
    Set c = AddCtl("Forms.ComboBox.1", l, t, w, ROW_H, , tip)
    c.Style = fmStyleDropDownList
    If IsArray(items) Then c.List = items
    Set Combo = c
End Function

' Multi-select list with a checkbox per row.
Private Function CheckList(ByVal l As Single, ByVal t As Single, ByVal w As Single, ByVal h As Single, _
                           ByVal tip As String) As MSForms.ListBox
    Dim c As MSForms.ListBox
    Set c = AddCtl("Forms.ListBox.1", l, t, w, h, , tip)
    c.ListStyle = fmListStyleOption
    c.MultiSelect = fmMultiSelectMulti
    Set CheckList = c
End Function

' Section header: bold caption with a thin line under it. Returns the top of the first row.
Private Function SectionHeader(ByVal cap As String, ByVal l As Single, ByVal t As Single, ByVal w As Single) As Single
    Dim hdr As MSForms.Label, ln As MSForms.Label
    Set hdr = AddCtl("Forms.Label.1", l, t, w, 12, cap)
    hdr.Font.Bold = True
    Set ln = AddCtl("Forms.Label.1", l, t + 13, w, 1)
    ln.BackStyle = fmBackStyleOpaque
    ln.BackColor = &H808080
    SectionHeader = t + 18
End Function

'=============================== BUILD THE FORM ================================
Private Sub UserForm_Initialize()
    Dim yLeft As Single, yRight As Single
    If swApp Is Nothing Then Set swApp = Application.SldWorks
    If gFormatCount = 0 Then LoadFormats
    gFormLoaded = True
    Set mDoc = swApp.ActiveDoc
    Set mKeys = New Collection
    If mDoc Is Nothing Then
        Me.Caption = "SWExporter  (batch - no document open)"
    Else
        Me.Caption = "SWExporter  -  " & mDoc.GetTitle
    End If

    yLeft = BuildWhatSection(6)
    yLeft = BuildScopeSection(yLeft + SECTION_GAP)
    yRight = BuildFormatsSection(6)
    yRight = BuildOutputSection(yRight + SECTION_GAP)
    yRight = BuildOptionsSection(yRight + SECTION_GAP)
    If yRight > yLeft Then yLeft = yRight
    BuildButtons yLeft + 10

    FillModes                       ' fills the item and format lists too
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)

        ' the X button during a run cancels instead of unloading the form mid-export

        gFormLoaded = False
    End If

'=================================== MODES =====================================
Private Function DocType() As Long
    If Not mDoc Is Nothing Then DocType = mDoc.GetType
End Function

Private Function ModeLabel(ByVal md As Long) As String
    Dim isAsm As Boolean
    isAsm = (DocType() = swDocumentTypes_e.swDocASSEMBLY)
    Select Case md
        Case MODE_BODIES: ModeLabel = "Each body"
        Case MODE_CFGBODIES: ModeLabel = "Each body x configuration"
        Case MODE_CONFIGS: ModeLabel = IIf(isAsm, "Whole assembly", "Whole part") & " (per configuration)"
        Case MODE_COMPONENTS: ModeLabel = "Each part"
        Case MODE_DISPSTATES: ModeLabel = "Display states"
        Case MODE_SHEETS: ModeLabel = "Each sheet"
        Case Else: ModeLabel = "Batch (folder or files)"
    End Select
End Function

' One-line description shown under the mode box (and as its tooltip).
Private Function ModeHelp(ByVal md As Long) As String
    Dim isAsm As Boolean
    isAsm = (DocType() = swDocumentTypes_e.swDocASSEMBLY)
    Select Case md
        Case MODE_BODIES
            If isAsm Then
                ModeHelp = "One file per ticked visible body. Hidden, suppressed and envelope components " & _
                           "and hidden bodies are skipped."
            Else
                ModeHelp = "One file per ticked body of the active configuration. Only visible bodies are " & _
                           "listed; tick Surface bodies to add surfaces."
            End If
        Case MODE_CFGBODIES
            ModeHelp = "One file per visible body, in every ticked configuration."
        Case MODE_CONFIGS
            ModeHelp = "One file of the whole " & IIf(isAsm, "assembly", "part") & " per ticked configuration."
        Case MODE_COMPONENTS
            ModeHelp = "One file per visible part (each part + configuration once). " & _
                       "Bodies hidden in the assembly are left out."
        Case MODE_DISPSTATES
            ModeHelp = "Every visible body of each ticked display state, one file each. Pick the " & _
                       "configurations in section 2. Repeats are skipped."
        Case MODE_SHEETS
            ModeHelp = "One file per ticked sheet. PDF: one file or one per sheet (see Formats)."
        Case Else
            ModeHelp = "Every ticked file in the list. Parts are split per 'Parts in batch'; " & _
                       "assemblies export each part inside."
    End Select
End Function

Private Sub FillModes()
    Dim ids As Variant, i As Long
    Select Case DocType()
        Case swDocumentTypes_e.swDocPART
            ids = Array(MODE_BODIES, MODE_CFGBODIES, MODE_CONFIGS, MODE_DISPSTATES, MODE_BATCH)
        Case swDocumentTypes_e.swDocASSEMBLY
            ids = Array(MODE_CONFIGS, MODE_COMPONENTS, MODE_BODIES, MODE_DISPSTATES, MODE_BATCH)
        Case swDocumentTypes_e.swDocDRAWING
            ids = Array(MODE_SHEETS, MODE_BATCH)
        Case Else
            ids = Array(MODE_BATCH)
    End Select
    ReDim mModeIds(0 To UBound(ids))
    cboMode.Clear
    For i = 0 To UBound(ids)
        mModeIds(i) = ids(i)
        cboMode.AddItem ModeLabel(mModeIds(i))
    Next i
    cboMode.ListIndex = 0           ' fires cboMode_Change
End Sub

Private Function CurMode() As Long
    If cboMode.ListIndex < 0 Then
        CurMode = MODE_BATCH
    Else
        CurMode = mModeIds(cboMode.ListIndex)
    End If
End Function

Private Sub cboMode_Change()
    Dim md As Long
    If cboMode.ListIndex < 0 Then Exit Sub
    md = CurMode()
    lblModeHelp.Caption = ModeHelp(md)
    cboMode.ControlTipText = ModeHelp(md)
    lblModeHelp.ControlTipText = ModeHelp(md)
    UpdateModeControls md
    FillItems
    FillFormats
End Sub

' Shows / enables only what the mode uses.
Private Sub UpdateModeControls(ByVal md As Long)
    Dim isBatch As Boolean, notSheets As Boolean
    isBatch = (md = MODE_BATCH)
    notSheets = (md <> MODE_SHEETS)
    chkSurf.Enabled = (md = MODE_BODIES Or md = MODE_CFGBODIES Or md = MODE_DISPSTATES Or isBatch)
    lblDsScope.Enabled = (md = MODE_DISPSTATES)
    cboDsScope.Enabled = lblDsScope.Enabled
    lblCfgFilter.Enabled = isBatch Or (md = MODE_DISPSTATES And DsAllCfgs())
    txtCfgFilter.Enabled = lblCfgFilter.Enabled
    ' batch source: only in Batch
    lblBatch.Visible = isBatch
    txtBatch.Visible = isBatch
    btnBrowseBatch.Visible = isBatch
    btnRefresh.Visible = isBatch
    btnAddFiles.Visible = isBatch
    chkRecurse.Visible = isBatch
    lblBatchModel.Visible = isBatch
    cboBatchModel.Visible = isBatch
    ' format options
    chkPdfOne.Enabled = (md = MODE_SHEETS Or isBatch)
    lblUnits.Enabled = notSheets
    cboUnits.Enabled = notSheets
    lblQuality.Enabled = notSheets
    cboQuality.Enabled = notSheets
    chkBinary.Enabled = notSheets
End Sub

Private Sub chkSurf_Click()
    If CurMode() = MODE_BODIES Then FillItems      ' body lists depend on it
End Sub

'================================= ITEM LIST ===================================
Private Sub ClearItems()
    lstItems.Clear
    Set mKeys = New Collection
End Sub

Private Sub AddListItem(ByVal text As String, ByVal key As String, ByVal ticked As Boolean)
    lstItems.AddItem text
    mKeys.Add key
    lstItems.Selected(lstItems.ListCount - 1) = ticked
End Sub

Private Sub AddItems(ByVal c As Collection, ByVal ticked As Boolean)
    Dim i As Long
    For i = 1 To c.Count
        AddListItem CStr(c(i)), CStr(c(i)), ticked
    Next i
End Sub

Private Sub FillItems()
    Dim md As Long
    md = CurMode()
    mFilling = True
    ClearItems
    lstItems.ControlTipText = TIP_ITEMS
    Select Case md
        Case MODE_BODIES
            If DocType() = swDocumentTypes_e.swDocASSEMBLY Then
                FillAssemblyBodyItems
            Else
                lblItems.Caption = "Visible bodies:"
                AddItems GetBodyNames(mDoc, CBool(chkSurf.Value)), True
            End If
        Case MODE_CONFIGS, MODE_CFGBODIES
            lblItems.Caption = IIf(md = MODE_CONFIGS, "Configurations:", "Configurations (all visible bodies of each):")
            FillConfigItems
        Case MODE_COMPONENTS
            lblItems.Caption = "Parts [configuration] (each once):"
            FillPartItems
        Case MODE_DISPSTATES
            FillDisplayStateItems
        Case MODE_SHEETS
            lblItems.Caption = "Sheets:"
            FillSheetItems
        Case MODE_BATCH
            lblItems.Caption = "Files:"
            FillBatchItems
    End Select
    mFilling = False
    UpdateCount
End Sub

' Assembly Each body: one row per body shown in the assembly; key = "path|config" & Tab & body.

    Dim d As Object, dc As Object, k As Variant, c As Collection, i As Long

        Next i

End Sub

Private Sub FillSheetItems()

Private Sub FillBatchItems()

        If mDoc.GetPathName <> "" Then
            txtBatch.Text = Left$(mDoc.GetPathName, InStrRev(mDoc.GetPathName, "\") - 1)

End Sub

' Display states: one row per state name.
'   Active configuration only: the states of the active configuration, all ticked.
'   All configurations: every state name found in the configurations matching Config filter
'   (active configuration's states first); names in the active configuration start ticked.
' When the list is rebuilt (scope / Config filter / Match case change) ticks are kept by name.
Private Sub FillDisplayStateItems()
    Dim act As String, actStates As Collection, names As Collection, cfgs As Collection
    Dim c As Collection, i As Long, j As Long, s As String, tick As Boolean
    act = mDoc.ConfigurationManager.ActiveConfiguration.Name
    Set actStates = ConfigDisplayStates(mDoc, act)
    Set names = New Collection
    If DsAllCfgs() Then
        lblItems.Caption = "Display states (all matching configurations):"
        lstItems.ControlTipText = "Each ticked state is exported in every matching configuration that has it."
        Set cfgs = ConfigsForDisplayStates(mDoc, Trim$(txtCfgFilter.Text), CBool(chkCase.Value))
        For i = 1 To cfgs.Count
            Set c = ConfigDisplayStates(mDoc, CStr(cfgs(i)))
            For j = 1 To c.Count
                AddUnique names, CStr(c(j))
            Next j
        Next i
    Else
        lblItems.Caption = "Display states (" & act & "):"
        lstItems.ControlTipText = "Each ticked state is exported in the active configuration '" & act & "'."
        For i = 1 To actStates.Count
            AddUnique names, CStr(actStates(i))
        Next i
    End If
    For i = 1 To names.Count
        s = CStr(names(i))
        If mDsPrevAll Is Nothing Then
            tick = InCollection(actStates, s)
        ElseIf InCollection(mDsPrevAll, s) Then
            tick = InCollection(mDsPrevTicked, s)
        Else
            tick = InCollection(actStates, s)
        End If
        AddListItem s, s, tick
    Next i
End Sub

' Rebuilds the display-state list keeping the ticks by state name.
Private Sub RefillDisplayStates()
    Dim i As Long
    If CurMode() <> MODE_DISPSTATES Then Exit Sub
    Set mDsPrevAll = New Collection
    Set mDsPrevTicked = New Collection
    For i = 0 To lstItems.ListCount - 1
        AddUnique mDsPrevAll, CStr(mKeys(i + 1))
        If lstItems.Selected(i) Then AddUnique mDsPrevTicked, CStr(mKeys(i + 1))
    Next i
    FillItems
    Set mDsPrevAll = Nothing
    Set mDsPrevTicked = Nothing
End Sub

Private Function DsAllCfgs() As Boolean
    If Not cboDsScope Is Nothing Then DsAllCfgs = (cboDsScope.ListIndex = 1)
End Function

Private Sub cboDsScope_Change()
    If mFilling Or cboMode Is Nothing Then Exit Sub
    If cboMode.ListIndex < 0 Then Exit Sub          ' still building the form
    UpdateModeControls CurMode()
    RefillDisplayStates
End Sub

' "3 of 8 ticked"; Display states adds the number of configuration x state visits.
Private Sub UpdateCount()
    Dim i As Long, n As Long, sel As Collection, plan As Collection, missing As Collection
    If mFilling Or lblCount Is Nothing Then Exit Sub
    Set sel = New Collection
    For i = 0 To lstItems.ListCount - 1
        If lstItems.Selected(i) Then
            n = n + 1
            sel.Add CStr(mKeys(i + 1))
        End If
    Next i
    lblCount.Caption = n & " of " & lstItems.ListCount & " ticked"
    If CurMode() = MODE_DISPSTATES And n > 0 Then
        Set missing = New Collection
        Set plan = DisplayStatePlan(mDoc, sel, DsAllCfgs(), Trim$(txtCfgFilter.Text), CBool(chkCase.Value), missing)
        lblCount.Caption = lblCount.Caption & vbLf & plan.Count & " state/config run(s)" & _
                           IIf(missing.Count > 0, ", " & missing.Count & " skipped", "")
    End If
End Sub

Private Sub lstItems_Change()
    UpdateCount
End Sub

Private Sub txtCfgFilter_Change()
    If CurMode() = MODE_DISPSTATES And DsAllCfgs() Then RefillDisplayStates
End Sub

'================================ FORMAT LIST ==================================
Private Sub FillFormats()
    Dim i As Long, n As Long, wantKind As String, prefix As String
    Select Case CurMode()
        Case MODE_SHEETS: wantKind = "DRAWING"
        Case MODE_BATCH: wantKind = ""
        Case Else: wantKind = "MODEL"
    End Select
    lstFormats.Clear
    ReDim mFmtIdx(0 To gFormatCount)
    For i = 0 To gFormatCount - 1
        If wantKind = "" Or gFormats(i).Kind = wantKind Then
            prefix = ""
            If wantKind = "" Then
                If gFormats(i).Kind = "MODEL" Then prefix = "Model: " Else prefix = "Drawing: "
            End If
            lstFormats.AddItem prefix & gFormats(i).Label & "  (" & gFormats(i).Ext & ")"
            mFmtIdx(n) = i
            lstFormats.Selected(n) = gFormats(i).DefaultOn
            n = n + 1
        End If
    Next i
End Sub

' Ticked item keys into gSelItems. False (with a message) if nothing is ticked.
Private Function CollectItems() As Boolean
    Dim i As Long
    Set gSelItems = New Collection
    For i = 0 To lstItems.ListCount - 1
        If lstItems.Selected(i) Then gSelItems.Add CStr(mKeys(i + 1))
    Next i
    If gSelItems.Count = 0 Then
        MsgBox "Tick at least one row in the list.", vbExclamation, "SWExporter"
        Exit Function
    End If
    CollectItems = True
End Function

' Ticked formats into gFormats().Selected. False (with a message) if none.
Private Function CollectFormats() As Boolean
    Dim i As Long
    For i = 0 To gFormatCount - 1
        gFormats(i).Selected = False
    Next i
    For i = 0 To lstFormats.ListCount - 1
        If lstFormats.Selected(i) Then
            gFormats(mFmtIdx(i)).Selected = True
            CollectFormats = True
        End If
    Next i
    If Not CollectFormats Then MsgBox "Tick at least one format.", vbExclamation, "SWExporter"
End Function

' Every other option into the g* globals. False (with a message) if something is missing.
Private Function CollectOptions() As Boolean
    gMode = CurMode()
    gIncludeSurfaces = CBool(chkSurf.Value)
    gOutMode = cboOut.ListIndex
    gOutText = Trim$(txtOut.Text)
    If gOutMode = 2 And gOutText = "" Then
        MsgBox "Choose a custom output folder.", vbExclamation, "SWExporter"
        Exit Function
    End If
    gTemplate = Trim$(txtTemplate.Text)
    gCollision = cboCollision.ListIndex
    gStlUnits = cboUnits.ListIndex                  ' 0..4 = swLengthUnit_e mm, cm, m, in, ft
    gStlQuality = Choose(cboQuality.ListIndex + 1, swSTLQuality_e.swSTLQuality_Coarse, _
                         swSTLQuality_e.swSTLQuality_Fine, swSTLQuality_e.swSTLQuality_Custom)
    gStlBinary = CBool(chkBinary.Value)
    gPdfCombined = CBool(chkPdfOne.Value)
    gDryRun = CBool(chkDry.Value)
    gWriteLog = CBool(chkLog.Value)
    gBatchFolder = Trim$(txtBatch.Text)
    gBatchModelMode = Choose(cboBatchModel.ListIndex + 1, MODE_CONFIGS, MODE_BODIES, MODE_CFGBODIES)
    gCfgFilter = Trim$(txtCfgFilter.Text)
    gCfgFilterCase = CBool(chkCase.Value)
    gDsAllCfgs = DsAllCfgs()
    CollectOptions = True
End Function

Private Sub btnExport_Click()
    If mBusy Then Exit Sub
    If Not CollectItems() Then Exit Sub
    If Not CollectFormats() Then Exit Sub
    If Not CollectOptions() Then Exit Sub

    If gMode = MODE_DISPSTATES Then
        If DisplayStatePlan(mDoc, gSelItems, gDsAllCfgs, gCfgFilter, gCfgFilterCase).Count = 0 Then
            MsgBox "None of the ticked display states exists in the chosen configurations.", _
                   vbExclamation, "SWExporter"
            Exit Sub
        End If
    End If
    If gMode = MODE_BATCH And Not gDryRun Then
        If MsgBox("Export " & gSelItems.Count & " file(s)? Files that are not already open will be " & _
                  "opened and closed again without saving.", vbOKCancel + vbQuestion, "SWExporter") <> vbOK Then Exit Sub
    End If

    mBusy = True
    btnExport.Enabled = False
    btnClose.Caption = "Cancel"
    RunExport mDoc
    mBusy = False
    btnExport.Enabled = True
    btnClose.Caption = "Close"
    ShowStatus "Done."
End Sub
